cmake_minimum_required(VERSION 3.22)
set(CMAKE_DEPENDS_USE_COMPILER OFF CACHE BOOL "Windows GCC depfiles contain drive colons" FORCE)
project(sc1 LANGUAGES C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(ROOT "${CMAKE_CURRENT_SOURCE_DIR}")
file(READ "${ROOT}/dependencies.json" SC1_LOCK)
string(JSON SC1_XBE_SHA256 GET "${SC1_LOCK}" executable sha256)
configure_file(src/xbe_identity.h.in "${CMAKE_CURRENT_BINARY_DIR}/xbe_identity.h" @ONLY)
set(XBOXRECOMP_DIR "${ROOT}/private/burnout3-sw/xboxrecomp")
if(NINTENDO_SWITCH)
  set(XBOXRECOMP_VULKAN ON CACHE BOOL "" FORCE)
else()
  set(XBOXRECOMP_VULKAN OFF CACHE BOOL "" FORCE)
endif()
set(NVK_SDK "${ROOT}/../BLACK/private/vulkan-sdk" CACHE PATH "Existing NVK SDK")
set(glslang_DIR "${ROOT}/../BLACK/private/glslang-sdk/lib/cmake/glslang" CACHE PATH "Existing Switch glslang")
if(NOT NINTENDO_SWITCH AND NOT WIN32)
  message(FATAL_ERROR "Use scripts/Build.ps1 for Switch, or scripts/Build-Host.cmd for diagnosis")
endif()
if(NOT EXISTS "${ROOT}/generated/recomp_funcs.h")
  message(FATAL_ERROR "Run scripts/generate.py first")
endif()
add_subdirectory("${XBOXRECOMP_DIR}" xboxrecomp)
get_target_property(KERNEL_SOURCES xbox_kernel SOURCES)
list(REMOVE_ITEM KERNEL_SOURCES kernel_bridge.c nv2a_pb_exec.c xbox_memory_layout.c)
list(APPEND KERNEL_SOURCES
  "${ROOT}/private/host/kernel_bridge.c"
  "${ROOT}/private/host/nv2a_pb_exec.c"
  "${ROOT}/private/host/xbox_memory_layout.c"
  "${ROOT}/src/gpu_commands.c"
  "${ROOT}/src/d3d_sync.c")
set_property(TARGET xbox_kernel PROPERTY SOURCES "${KERNEL_SOURCES}")
if(NINTENDO_SWITCH)
file(SHA256 "${NVK_SDK}/lib/libvulkan.a" SC1_DRIVER_SHA)
string(JSON SC1_EXPECTED_DRIVER GET "${SC1_LOCK}" horizon_sdk nvk_sha256)
if(NOT SC1_DRIVER_SHA STREQUAL SC1_EXPECTED_DRIVER)
  message(FATAL_ERROR "Expected the existing self-contained Horizon NVK SDK archive")
endif()
get_target_property(NVK_SOURCES xbox_nv2a_vk SOURCES)
list(REMOVE_ITEM NVK_SOURCES nv2a_vk.c)
list(APPEND NVK_SOURCES "${ROOT}/private/host/nv2a_vk.c")
set_property(TARGET xbox_nv2a_vk PROPERTY SOURCES "${NVK_SOURCES}")
# The local NVK SDK is the self-contained archive already linked by BLACK.
# It bundles Expat/zlib/Rust; the upstream raw Mesa SDK instead needs portlibs.
get_target_property(NVK_LINK xbox_nv2a_vk INTERFACE_LINK_LIBRARIES)
foreach(NAME expat z)
  find_library(SC1_${NAME}_LIB NAMES ${NAME} PATHS "$ENV{DEVKITPRO}/portlibs/switch/lib")
  if(NOT SC1_${NAME}_LIB)
    list(REMOVE_ITEM NVK_LINK ${NAME})
  endif()
endforeach()
set_property(TARGET xbox_nv2a_vk PROPERTY INTERFACE_LINK_LIBRARIES "${NVK_LINK}")
endif()
file(GLOB GENERATED CONFIGURE_DEPENDS "${ROOT}/generated/*.c")
add_executable(sc1 private/host/main.c private/host/switch_nx.c
  private/host/switch_saltynx.c src/recomp_manual.c src/cpu.c src/xbe_identity.c src/guest_exceptions.c src/dsp_commands.c ${GENERATED})
target_include_directories(sc1 PRIVATE "${ROOT}/generated" "${XBOXRECOMP_DIR}/include"
  "${XBOXRECOMP_DIR}/src/apu" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(sc1 PRIVATE xboxrecomp)
if(WIN32)
  option(SC1_ABI_CHECK "Log guest register/stack contract violations" OFF)
  if(SC1_ABI_CHECK)
    target_compile_definitions(sc1 PRIVATE RECOMP_ABI_CHECK=1)
    target_compile_definitions(xbox_kernel PRIVATE RECOMP_ABI_CHECK=1)
  endif()
  target_sources(sc1 PRIVATE src/host_threads.c src/host_scanout.c)
  # clang-cl already has these builtins; rename the fork's MSVC wrappers.
  foreach(LIB sc1 xbox_apu xbox_nv2a)
    target_compile_definitions(${LIB} PRIVATE __builtin_ctz=sc1_builtin_ctz __builtin_clz=sc1_builtin_clz)
  endforeach()
  target_compile_options(xbox_kernel PRIVATE "/FI${ROOT}/src/host_clock.h")
  target_compile_options(xbox_usb PRIVATE "/FI${ROOT}/src/host_clock.h")
  target_link_libraries(sc1 PRIVATE d3d11 dxgi dxguid xinput winmm dbghelp)
  set_source_files_properties(${GENERATED} PROPERTIES COMPILE_OPTIONS "/Od;/w;/bigobj")
  target_link_options(sc1 PRIVATE /LARGEADDRESSAWARE /DEBUG /STACK:8388608)
else()
  target_compile_definitions(sc1 PRIVATE NFSU2_VULKAN=1)
  set(SC1_GEN_OPT "-O1" CACHE STRING "Optimisation of generated game code; -O0 for bring-up")
  set_source_files_properties(${GENERATED} PROPERTIES
    COMPILE_OPTIONS "${SC1_GEN_OPT};-w;-fno-strict-aliasing;-fwrapv")
  target_link_libraries(sc1 PRIVATE nx m)
target_link_options(sc1 PRIVATE "LINKER:--wrap=SDL_SYS_SetThreadPriority"
  "LINKER:--wrap=audrvVoiceAddWaveBuf")
nx_generate_nacp(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/sc1.nacp"
  NAME "Splinter Cell - development" AUTHOR "SC1 recomp contributors" VERSION "0.0.1")
nx_create_nro(sc1 NACP "${CMAKE_CURRENT_BINARY_DIR}/sc1.nacp")
endif()
